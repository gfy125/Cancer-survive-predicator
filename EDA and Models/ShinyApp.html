# Define server logic
server <- function(input, output, session) {
  # Extract non-zero coefficients
  coef_df <- as.data.frame(as.matrix(coef(cv_fit, s = lambda_min)))
  coef_df$feature <- rownames(coef_df)
  colnames(coef_df)[1] <- "coefficient"
  selected_features <- coef_df$feature[coef_df$coefficient != 0]

  # Separate genes from clinical variables
  selected_genes <- selected_features[grepl("^ENSG", selected_features)]
  selected_clinical <- selected_features[!grepl("^ENSG", selected_features)]

  output$geneInputs <- renderUI({
    lapply(selected_genes, function(gene) {
      numericInput(gene, label = gene, value = 0)
    })
  })

  predictRisk <- reactive({
    req(input$age)

    # Construct patient data
    new_patient <- as.list(setNames(rep(0, length(selected_features)), selected_features))

    # Clinical features
    new_patient[["age_diagnosis"]] <- input$age

    # Gender dummy
    if ("gendermale" %in% selected_features) new_patient[["gendermale"]] <- ifelse(input$gender == "gendermale", 1, 0)
    if ("genderfemale" %in% selected_features) new_patient[["genderfemale"]] <- ifelse(input$gender == "genderfemale", 1, 0)

    # Vital status dummy
    if ("vital_statusAlive" %in% selected_features) new_patient[["vital_statusAlive"]] <- ifelse(input$vital == "vital_statusAlive", 1, 0)
    if ("vital_statusDead" %in% selected_features) new_patient[["vital_statusDead"]] <- ifelse(input$vital == "vital_statusDead", 1, 0)

    # AJCC stage dummies
    for (s in levels(factor(input$stage))) {
      key <- paste0("ajcc_pathologic_stage", s)
      if (key %in% selected_features) {
        new_patient[[key]] <- ifelse(input$stage == s, 1, 0)
      }
    }

    # Gene expressions
    for (gene in selected_genes) {
      val <- input[[gene]]
      if (!is.null(val)) new_patient[[gene]] <- val
    }

    # Convert to matrix
    patient_mat <- matrix(unlist(new_patient), nrow = 1)
    colnames(patient_mat) <- names(new_patient)

    # Predict
    predict(cv_fit, newx = patient_mat, s = lambda_min, type = "link")[1]
  })

  output$riskScore <- renderPrint({
    score <- predictRisk()
    cat("Linear Risk Score:", round(score, 4))
  })

  output$survPlot <- renderPlot({
    lp <- predictRisk()
    # Create baseline survival from training set
    y <- cv_fit$y  # extract the survival object if available
    if (is.null(y)) {
      y <- Surv(rep(1, 100), rep(1, 100))  # fallback dummy
    }

    fit_km <- survfit(y ~ 1)
    surv_probs <- fit_km$surv ^ exp(lp)
    time_years <- fit_km$time

    plot(time_years, surv_probs, type = "l", lwd = 2,
         xlab = "Time (Years)", ylab = "Survival Probability",
         main = "Predicted Survival Curve")
  })
}
